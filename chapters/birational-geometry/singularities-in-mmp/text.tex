\subsection{Discrepancies and log discrepancies}

  \begin{definition}
    Let $E$ be a prime divisor over $X$.
    Let $f: Y \to X$ be a log resolution of $(X,B)$ such that $E$ is a prime divisor on $Y$.
    Write
    \[ K_Y + B_Y = f^*(K_X + B) + \sum_E a(E,X,B) E, \]
    where $B_Y$ is the strict transform of $B$.
    Then $a(E,X,B)$ is called the discrepancy of $E$ with respect to $(X,B)$ and $a(E,X,B) + 1$ is called the log discrepancy of $E$ with respect to $(X,B)$.
    \Yang{}
  \end{definition}
